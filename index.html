  function speakText() {
  let text = document.getElementById("output").innerText;

  if (!text) {
    alert("Generate a story first!");
    return;
  }

  let speech = new SpeechSynthesisUtterance();
  speech.text = text;
  speech.lang = "en-US";
  speech.rate = 1;
  speech.pitch = 1;

  window.speechSynthesis.speak(speech);
}

  
  

  

  function generateImages() {
  let container = document.getElementById("images");
  container.innerHTML = "";

  let scenes = [
    "Pixar 3D rabbit walking in rainy forest cinematic lighting",
    "Pixar 3D rabbit discovering injured bird emotional scene",
    "Pixar 3D rabbit facing dark predator dramatic lighting",
    "Pixar 3D emotional sunrise hope ending cinematic"
  ];

  scenes.forEach((scene, index) => {
    let imgUrl = "https://image.pollinations.ai/prompt/" + encodeURIComponent(scene);

    let img = document.createElement("img");
    img.src = imgUrl;
    img.style.width = "250px";
    img.style.margin = "10px";
    img.style.borderRadius = "10px";

    let title = document.createElement("p");
    title.innerText = "Scene " + (index + 1);

    container.appendChild(title);
    container.appendChild(img);
  });
}
